cmake_minimum_required(VERSION 3.16)
project(Vaa3DSharedCoreTests LANGUAGES CXX)
# Small, dependency-free regression gate. This is NOT a full application build.
enable_testing()
find_package(Threads REQUIRED)
foreach(name maximum_z_projection rescale_one_micron)
    add_executable(${name}_test ${name}_test.cpp)
    target_compile_features(${name}_test PRIVATE cxx_std_11)
    target_link_libraries(${name}_test PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name}_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
